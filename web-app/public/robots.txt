# See https://www.robotstxt.org/robotstxt.html for documentation on how to use the robots.txt file

User-agent: *
Disallow: /my/lists
Disallow: /user_lists
Disallow: /searches

# Filter URLs combining two or more categories or countries. FilterPath emits a
# comma only for a multi-valued segment, so the comma is exactly that boundary.
# Both rules lead with /* because robots.txt matching is prefix-based and an
# /rc/<id>/ prefix would otherwise slip past them.
Disallow: /*the-greatest/*,*
Disallow: /*written-by/*,*

# The correction form. Nothing here is content, and it is the surface that has
# been used to flood the origin. One rule for all three domains AND both URL
# forms: the public route sits inside `scope "(/rc/:ranking_configuration_id)"`,
# so e.g. /book/*/suggest-correction alone would miss the live
# /rc/:id/book/*/suggest-correction form. /*/suggest-correction, unanchored,
# matches both.
Disallow: /*/suggest-correction

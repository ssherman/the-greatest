User-agent: *
Disallow: /my/lists
Disallow: /user_lists
Disallow: /searches

# Filter URLs combining two or more categories or countries. FilterPath emits a
# comma only for a multi-valued segment, so the comma is exactly that boundary.
# Both rules lead with /* because robots.txt matching is prefix-based and an
# /rc/<id>/ prefix would otherwise slip past them.
Disallow: /*the-greatest/*,*
Disallow: /*written-by/*,*
